{%- comment -%} Article structured data for paper summaries. Included from head.html, so the rendered post body is available as `content`. Every value goes through `jsonify` so quotes or special characters can never break the JSON. - headline: the paper title, or `seo_title` when the title exceeds Google's 110-character headline limit. - image: `schema_image` front matter if set, else the first image in the post body, else the site logo. - dateModified: `last_modified_at` front matter if set, else the post date. - keywords: the glossary terms the post links to. - articleSection: display names of the post's themes. {%- endcomment -%} {%- assign page_url = page.url | prepend: site.baseurl | prepend: site.url -%} {%- assign headline = page.title -%} {%- if headline.size > 110 -%} {%- assign headline = page.seo_title | default: page.title | truncate: 110 -%} {%- endif -%} {%- assign schema_image = page.schema_image -%} {%- unless schema_image -%} {%- assign img_parts = content | split: ' { "@context": "https://schema.org", "@type": "Article", "headline": {{ headline | jsonify }}, {%- if headline != page.title %} "alternativeHeadline": {{ page.title | jsonify }}, {%- endif %} "image": [{{ schema_image | jsonify }}], "datePublished": {{ page.date | date_to_xmlschema | jsonify }}, "dateModified": {{ modified | date_to_xmlschema | jsonify }}, "author": { "@type": "Person", "name": {{ site.author | jsonify }}, "url": {{ site.url | append: site.baseurl | append: '/about/' | jsonify }} }, "publisher": { "@type": "Organization", "name": {{ site.title | jsonify }}, "url": {{ site.url | append: site.baseurl | append: '/' | jsonify }}, "logo": { "@type": "ImageObject", "url": {{ site.logo | prepend: site.baseurl | prepend: site.url | jsonify }} } }, {%- if page.meta_description %} "description": {{ page.meta_description | strip_html | strip_newlines | jsonify }}, {%- endif %} "mainEntityOfPage": { "@type": "WebPage", "@id": {{ page_url | jsonify }} }, {%- if sections.size > 0 %} "articleSection": {{ sections | jsonify }}, {%- endif %} {%- if keywords.size > 0 %} "keywords": {{ keywords | jsonify }}, {%- endif %} {%- if page.doi %} "citation": { "@type": "ScholarlyArticle", "name": {{ page.title | jsonify }}, "identifier": {{ page.doi | jsonify }}, "url": {{ page.doi | prepend: 'https://doi.org/' | jsonify }} }, {%- endif %} {%- if page.snippet %} "abstract": {{ page.snippet | strip_html | strip_newlines | jsonify }}, {%- endif %} "isAccessibleForFree": true }
